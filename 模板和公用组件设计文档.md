## 一、设计理念

### 1.1 核心原则
```
✅ 一致性优先：所有页面保持统一的视觉风格和交互模式
✅ 响应式布局：适配不同屏幕尺寸
✅ 渐进增强：基础功能优先，高级功能逐步增强
✅ 用户体验：减少用户输入，提供智能提示和自动补全
✅ 错误友好：完善的数据校验和错误提示机制
```

### 1.2 技术栈
- **前端框架**: Bootstrap 5.3.0
- **CSS预处理**: 内联样式（便于维护和定制）
- **JavaScript**: 原生JS（避免额外依赖）
- **后端框架**: Django 5.2.8
- **表单处理**: Django Forms
- **Excel处理**: openpyxl

```python
class FeatureForm(StripWhitespaceMixin, forms.Form):
    """功能表单"""
    
    # 1. 基础信息字段
    dynamic_id = forms.CharField(
        label="动态编号（如变更单号）",
        max_length=100,
        widget=forms.TextInput(attrs={'class': 'form-control'})
    )
    
    ops_remark = forms.CharField(
        label="操作备注（支持解析ONES链接）",
        max_length=200,
        required=False,
        widget=forms.TextInput(attrs={
            'class': 'form-control', 
            'placeholder': '输入任务描述或ONES链接'
        })
    )
    
    # 2. 单条记录字段
    single_field = forms.CharField(
        label="字段标签",
        required=False,
        widget=forms.TextInput(attrs={'class': 'form-control'})
    )
    
    # 3. Excel批量导入
    excel_file = forms.FileField(
        label="或上传Excel",
        required=False,
        help_text=".xlsx格式；列：字段1、字段2...",
        widget=forms.ClearableFileInput(attrs={'class': 'form-control'})
    )
    
    def clean(self):
        """表单级校验"""
        cleaned_data = super().clean()
        excel = cleaned_data.get('excel_file')
        single = cleaned_data.get('single_field')
        
        # 互斥校验
        if not excel and not single:
            raise forms.ValidationError("请至少填写单条记录或上传Excel文件")
        if excel and single:
            raise forms.ValidationError("不能同时填写单条记录和上传Excel文件")
        
        return cleaned_data
```

### 6.2 StripWhitespaceMixin

**作用**: 自动去除所有字符串字段的首尾空格

```python
class StripWhitespaceMixin:
    def clean(self):
        cleaned_data = super().clean()
        for k, v in list(cleaned_data.items()):
            if isinstance(v, str):
                field = self.fields.get(k)
                if field and isinstance(field.widget, forms.Textarea):
                    cleaned_data[k] = v.strip()
                else:
                    # 单行文本去除换行符
                    cleaned_data[k] = v.strip().replace("\r", "").replace("\n", "")
        return cleaned_data
```

```python
def validate_required(value, field_name):
    """必填项校验"""
    if value is None or (isinstance(value, str) and not value.strip()):
        return f"{field_name}不能为空"
    return None
```

### 7.2 校验流程

```python
def validate_records(records):
    """批量校验记录"""
    results = []
    passed_count = 0
    failed_count = 0
    
    for idx, record in enumerate(records):
        row_number = idx + 2  # 跳过表头
        errors = []
        
        # 必填项校验
        error = validate_required(record.get('field1'), '字段1')
        if error:
            errors.append(error)
        
        # 数值校验
        error = validate_number(record.get('field2'), '字段2')
        if error:
            errors.append(error)
        
        # 记录结果
        if errors:
            results.append({
                'row_number': row_number,
                'valid': False,
                'errors': errors
            })
            failed_count += 1
        else:
            results.append({
                'row_number': row_number,
                'valid': True,
                'errors': []
            })
            passed_count += 1
    
    return {
        'valid': failed_count == 0,
        'total': len(records),
        'passed': passed_count,
        'failed': failed_count,
        'results': results
    }
```

```python
def generate_validation_failure_excel(uploaded_file, validation_result, module_name):
    """生成包含校验错误的Excel"""
    uploaded_file.seek(0)
    wb = openpyxl.load_workbook(uploaded_file, data_only=True)
    ws = wb.active
    
    # 添加校验结果列
    max_col = ws.max_column
    ws.cell(row=1, column=max_col + 1, value='校验结果')
    
    # 填充错误信息
    for result in validation_result.get('results', []):
        row_num = result.get('row_number')
        if not result.get('valid'):
            errors = result.get('errors', [])
            error_msg = '; '.join(errors)
            ws.cell(row=row_num, column=max_col + 1, value=error_msg)
    
    # 保存临时文件
    temp_path = save_validation_temp_file(wb, module_name)
    return temp_path
```

### 8.1 自动填充（物资编码示例）

```javascript
function bindAutoFill(prefix) {
  const idInput = document.getElementById("id_" + prefix + "_item_id");
  const nameInput = document.getElementById("id_" + prefix + "_item_name");
  const uomInput = document.getElementById("id_" + prefix + "_item_uom");
  
  async function lookup() {
    const v = (idInput.value || "").trim();
    if (!v) {
      nameInput.value = "";
      uomInput.value = "";
      return;
    }
    
    try {
      const r = await fetch("/item/by-id/?id=" + encodeURIComponent(v));
      const j = await r.json();
      if (j.found) {
        nameInput.value = j.name || "";
        uomInput.value = j.uom || "";
      } else {
        // 显示错误提示
        setMsg("未找到该物资编码，请补充信息");
      }
    } catch (e) {
      setMsg("查询失败");
    }
  }
  
  idInput.addEventListener("blur", lookup);
  idInput.addEventListener("change", lookup);
}

// 绑定新物资和原物资

bindAutoFill("orig");
```

```python
"""
功能模块视图
"""
import io
import logging
from django.shortcuts import render
from django.http import FileResponse
from ..forms import FeatureForm
from ..navigation import SIDEBAR_GROUPS
from ..config import get_config
from ..sql_merge import chunk_list, format_in, merge_by_key
from ..logger_utils import log_view_input
from .base import parse_ops_remark, save_sql_file
from ..validation_utils import (
    validate_required,
    generate_validation_failure_excel,
    get_temp_filename_from_path,
)

try:
    import openpyxl
    from openpyxl.styles import Font, Alignment, PatternFill
    OPENPYXL_AVAILABLE = True
except ImportError:
    OPENPYXL_AVAILABLE = False

logger = logging.getLogger('work_tools.view')
sql_logger = logging.getLogger('work_tools.sql')


def validate_records(records):
    """校验记录"""
    results = []
    passed_count = 0
    failed_count = 0
    
    for idx, record in enumerate(records):
        row_number = idx + 2
        errors = []
        
        # 必填项校验
        error = validate_required(record.get('field1'), '字段1')
        if error:
            errors.append(error)
        
        if errors:
            results.append({
                'row_number': row_number,
                'valid': False,
                'errors': errors
            })
            failed_count += 1
        else:
            results.append({
                'row_number': row_number,
                'valid': True,
                'errors': []
            })
            passed_count += 1
    
    return {
        'valid': failed_count == 0,
        'total': len(records),
        'passed': passed_count,
        'failed': failed_count,
        'results': results
    }


def parse_excel(file):
    """解析Excel文件"""
    if not OPENPYXL_AVAILABLE:
        raise ValueError("缺少 openpyxl，无法解析 Excel")
    
    wb = openpyxl.load_workbook(file)
    ws = wb.active
    headers = [cell.value for cell in ws[1]]
    idx = {str(h): i for i, h in enumerate(headers)}
    
    def pick(alts):
        for a in alts:
            if a in idx:
                return idx[a]
        return None
    
    field1_idx = pick(['字段1', 'field1', 'FIELD1'])
    
    if field1_idx is None:
        raise ValueError("Excel 缺少必要列：字段1")
    
    records = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if not row:
            continue
        rec = {
            'field1': str(row[field1_idx]).strip() if row[field1_idx] else None,
        }
        records.append(rec)
    
    if not records:
        raise ValueError("Excel 中没有有效的行数据")
    
    logger.info(f"[功能] Excel解析完成, 记录数={len(records)}")
    return records


def generate_sql(records, ops_remark=None):
    """生成SQL"""
    cfg = get_config()
    sql = []
    sql.append("1、执行语句")
    
    sql_logger.info(f"[SQL合并] 开始生成SQL, 记录数={len(records)}")
    
    if cfg.get('MERGE_MODULES', {}).get('module_name', True):
        # SQL合并模式
        ids = [r['field1'] for r in records]
        for chunk in chunk_list(ids, int(cfg.get('MERGE_MAX_IN_SIZE', 500))):
            inlist = format_in(chunk)
            stmt = f"UPDATE table SET ... WHERE id IN ({inlist});"
            sql.append(stmt)
    else:
        # 逐条模式
        for r in records:
            stmt = f"UPDATE table SET ... WHERE id='{r['field1']}';"
            sql.append(stmt)
    
    sql.append("")
    sql.append("2、回退语句")
    # ... 回退语句生成
    
    return "\n".join(sql)


def feature_view(request):
    """功能视图"""
    saved_file = None
    validation_failure = None

    if request.method == 'POST':
        form = FeatureForm(request.POST, request.FILES)
        if form.is_valid():
            try:
                cd = form.cleaned_data
                ops_remark = parse_ops_remark(cd.get('ops_remark', ''))
                dynamic_id = cd.get('dynamic_id', '')
                
                # 记录视图输入
                log_view_input('feature', cd)
                
                logger.info(f"[功能] 开始处理, dynamic_id={dynamic_id}")
                
                # 解析数据
                if cd.get('excel_file'):
                    records = parse_excel(cd['excel_file'])
                    
                    # 执行数据校验
                    validation_result = validate_records(records)
                    
                    if not validation_result['valid']:
                        # 校验失败
                        temp_file = generate_validation_failure_excel(
                            cd['excel_file'], 
                            validation_result, 
                            'module_name'
                        )
                        
                        if temp_file:
                            validation_failure = {
                                'total': validation_result['total'],
                                'passed': validation_result['passed'],
                                'failed': validation_result['failed'],
                                'filename': get_temp_filename_from_path(temp_file)
                            }
                            logger.info(f"校验失败: 总行数={validation_failure['total']}, "
                                      f"失败行数={validation_failure['failed']}")
                        
                        return render(request, 'template.html', {
                            'form': form,
                            'validation_failure': validation_failure,
                            'active_menu': 'menu_name',
                            'sidebar_groups': SIDEBAR_GROUPS,
                        })
                else:
                    # 单条记录
                    rec = {
                        'field1': cd.get('field1'),
                    }
                    records = [rec]
                    logger.info(f"[功能] 单条模式, field1={rec['field1']}")
                
                # 生成SQL
                sql_content = generate_sql(records, ops_remark)
                logger.info(f"[功能] SQL生成完成")
                
                # 保存SQL文件
                saved_file = save_sql_file(
                    sql_content, prefix='功能', dynamic_id=dynamic_id)
                logger.info(f"[功能] SQL文件已保存: {saved_file}")
                
            except Exception as e:
                logger.error(f"[功能] 处理失败: {e}", exc_info=True)
                form.add_error(None, f"生成SQL失败: {str(e)}")
    else:
        form = FeatureForm()

    return render(request, 'template.html', {
        'form': form,
        'saved_file': saved_file,
        'validation_failure': validation_failure,
        'active_menu': 'menu_name',
        'sidebar_groups': SIDEBAR_GROUPS,
    })


def download_template(request):
    """下载Excel模板"""
    if not OPENPYXL_AVAILABLE:
        return FileResponse(
            io.BytesIO(b"openpyxl not installed"),
            as_attachment=True,
            filename="error.txt"
        )
    
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "模板"
    
    # 设置表头
    headers = ["字段1", "字段2", "字段3"]
    header_fill = PatternFill(start_color="4472C4",
                              end_color="4472C4", fill_type="solid")
    header_font = Font(bold=True, color="FFFFFF")
    
    for col_num, header in enumerate(headers, 1):
        cell = ws.cell(row=1, column=col_num)
        cell.value = header
        cell.fill = header_fill
        cell.font = header_font
        cell.alignment = Alignment(horizontal='center', vertical='center')
    
    # 设置列宽
    ws.column_dimensions['A'].width = 20
    ws.column_dimensions['B'].width = 20
    ws.column_dimensions['C'].width = 20
    
    # 添加示例数据
    example_data = [
        ["示例值1", "示例值2", "示例值3"],
    ]
    
    for row_num, row_data in enumerate(example_data, 2):
        for col_num, value in enumerate(row_data, 1):
            cell = ws.cell(row=row_num, column=col_num)
            cell.value = value
            cell.alignment = Alignment(horizontal='left', vertical='center')
    
    bio = io.BytesIO()
    wb.save(bio)
    bio.seek(0)
    
    response = FileResponse(bio, as_attachment=True,
                            filename='template.xlsx')
    response['Content-Type'] = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
    return response


__all__ = [
    'feature_view',
    'download_template',
]
```

## 十、最佳实践总结

### 10.1 开发流程
```
1. 定义Form类（forms.py）
   ├── 继承StripWhitespaceMixin
   ├── 定义所有字段
   └── 添加clean方法校验

2. 创建HTML模板
   ├── 复制标准模板
   ├── 修改标题和字段
   └── 调整分段结构

3. 实现视图函数
   ├── Excel解析函数
   ├── 数据校验函数
   ├── SQL生成函数
   └── 主视图函数

4. 配置URL路由
   └── urls.py添加路由

5. 更新导航菜单
   └── navigation.py添加菜单项
```

### 10.2 代码复用建议
- **样式**: 直接复制标准CSS，确保一致性
- **组件**: 使用 `{% include 'sidebar.html' %}` 引入侧边栏
- **工具函数**: 复用 `base.py` 中的工具函数
- **校验**: 使用 `validation_utils.py` 的标准校验函数
- **SQL合并**: 使用 `sql_merge.py` 的合并策略

### 10.3 注意事项
1. **CSRF保护**: 表单必须包含 `{% csrf_token %}`
2. **文件上传**: form标签必须设置 `enctype="multipart/form-data"`
3. **context变量**: 必须传递 `active_menu` 和 `sidebar_groups`
4. **日志记录**: 关键操作使用logger记录
5. **异常处理**: 捕获并友好提示用户
6. **会话管理**: 支持清除后恢复表单数据

### 10.4 性能优化
- SQL合并策略减少语句数量
- 防抖处理避免频繁搜索
- 分块处理大数据集
- 临时文件定期清理
